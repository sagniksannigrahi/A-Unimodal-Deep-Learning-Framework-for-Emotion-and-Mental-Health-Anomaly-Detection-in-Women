train.py

import os
import numpy as np
from sklearn.preprocessing import LabelEncoder
from sklearn.utils.class_weight import compute_class_weight

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import (
    Conv2D, MaxPooling2D,
    Flatten, Dense, Dropout
)
from tensorflow.keras.utils import to_categorical
from tensorflow.keras.callbacks import ReduceLROnPlateau, EarlyStopping

from preprocess import load_dataset

# ---------------- PATH ----------------
BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
MODEL_PATH = os.path.join(BASE_DIR, "models", "emotion_model.keras")
LABELS_PATH = os.path.join(BASE_DIR, "models", "label_classes.npy")

# ---------------- LOAD DATA ----------------
print("Loading dataset...")
X, y = load_dataset()

# Normalize
X = X / (np.max(X) + 1e-8)

# Encode labels
encoder = LabelEncoder()
y_int = encoder.fit_transform(y)
y_encoded = to_categorical(y_int)

np.save(LABELS_PATH, encoder.classes_)

# Add channel dimension
X = X[..., np.newaxis]

# ---------------- CLASS WEIGHTS ----------------
class_weights = compute_class_weight(
    class_weight='balanced',
    classes=np.unique(y_int),
    y=y_int
)

class_weights = dict(enumerate(class_weights))

# ---------------- MODEL ----------------
print("Building CNN model...")

model = Sequential()

model.add(Conv2D(32, (3,3), activation='relu', input_shape=(40,130,1)))
model.add(MaxPooling2D((2,2)))

model.add(Conv2D(64, (3,3), activation='relu'))
model.add(MaxPooling2D((2,2)))

model.add(Conv2D(128, (3,3), activation='relu'))
model.add(MaxPooling2D((2,2)))

model.add(Flatten())

model.add(Dense(128, activation='relu'))
model.add(Dropout(0.5))

model.add(Dense(y_encoded.shape[1], activation='softmax'))

# Compile
model.compile(
    optimizer='adam',
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

model.summary()

# ---------------- CALLBACKS ----------------
lr_scheduler = ReduceLROnPlateau(
    monitor='val_loss',
    factor=0.5,
    patience=5,
    min_lr=1e-5
)

early_stop = EarlyStopping(
    monitor='val_loss',
    patience=8,
    restore_best_weights=True
)

# ---------------- TRAIN ----------------
print("Training started...")

model.fit(
    X, y_encoded,
    epochs=50,
    batch_size=32,
    validation_split=0.2,
    class_weight=class_weights,
    callbacks=[lr_scheduler, early_stop],
    verbose=1
)

# ---------------- SAVE ----------------
model.save(MODEL_PATH)

print("\n✅ Training complete.")
